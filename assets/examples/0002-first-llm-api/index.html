# 0002 — First LLM call: messages, streaming, tools, structured output

One LangChain model handle (`ChatDeepSeek`), four stages. The point is the **shape of
the conversation**, not the vendor: every stage is messages in, messages out.

| stage | what it shows | stable result |
|---|---|---|
| 1. chat | `SystemMessage` + `HumanMessage` → `AIMessage`; turn 2 re-sends the whole history | 2 messages, then **4** |
| 2. streaming | `model.stream()` yields `AIMessageChunk` pieces; join them and you get the `invoke()` text | join == `invoke()`; the final chunk is often empty |
| 3. tools | `bind_tools()` → `AIMessage.tool_calls` → your code runs the tool → `ToolMessage` → one more call | thread grows **4 → 5** (one round trip) |
| 4. structured | `with_structured_output(SupportTicket, method="function_calling")` returns a validated pydantic instance | a `SupportTicket` object, not a dict |

Stage 3 is **one round trip, not an agent**. The model asks; your code executes; the
model answers. An agent is this loop repeated until no tool calls remain — that is
Lesson 3, by hand.

Model *wording* varies between runs. The counts above are structural (messages you
constructed), and `evals.py` asserts the contract, not the sentences.

## DeepSeek notes

- `deepseek-flash` is the current model name (DeepSeek-V4.1-Flash-1017) and supports
  tool calls and JSON output; `deepseek-v4-pro` is the larger option.
- The API is OpenAI-compatible, which is why `langchain-deepseek` can reuse the
  OpenAI transport. The key is `DEEPSEEK_API_KEY`.
- DeepSeek thinks by default; the lab passes
  `extra_body={"thinking": {"type": "disabled"}}` because this lesson is about
  plumbing. Remove it to watch the reasoning tokens (and pay for them).
- DeepSeek has no OpenAI-style `json_schema` response format, so structured output
  uses tool calling: `with_structured_output(SupportTicket, method="function_calling")`

langchain_lab.py

"""0002-first-llm-api - chat, streaming, tools, and structured output with LangChain.

One model handle, four stages:

  1. chat       SystemMessage + HumanMessage -> AIMessage, and a second turn where
                the whole history is re-sent
  2. streaming  model.stream() -> AIMessageChunk pieces; join them and you have
                exactly what invoke() returned
  3. tools      bind_tools() -> AIMessage.tool_calls -> your code runs the tool ->
                ToolMessage -> one more call. ONE round trip, not an agent loop.
  4. structured model.with_structured_output(SupportTicket) -> a validated pydantic
                instance, via DeepSeek's tool calling

This lab talks to the live DeepSeek API. Set DEEPSEEK_API_KEY first:

  PowerShell (this session):  $env:DEEPSEEK_API_KEY = "sk-..."
  PowerShell (persist):       setx DEEPSEEK_API_KEY "sk-..."   # reopen the terminal

Get a key at https://platform.deepseek.com/api_keys. No key -> exit 1 with these
instructions, never a guess.
"""

from __future__ import annotations

import os
import sys
from typing import Any, Literal

from langchain_core.messages import (
    AIMessage,
    BaseMessage,
    HumanMessage,
    SystemMessage,
    ToolMessage,
)
from langchain_core.tools import BaseTool, tool
from pydantic import BaseModel, Field

MODEL = os.getenv("DEEPSEEK_MODEL", "deepseek-flash")
BASE_URL = os.getenv("DEEPSEEK_BASE_URL", "https://api.deepseek.com")

SYSTEM = "You are a terse support assistant. Answer in one sentence."
QUESTION_1 = "My invoice charged me twice. What should I do?"
QUESTION_2 = "And how long does the refund take?"
QUESTION_3 = "Where is order A-1042?"
TICKET_TEXT = "My invoice charged me twice for order A-1042. Fix it now."

# --- the world the tool can touch (fixture) --------------------------------

ORDERS = {
    "A-1042": "shipped from the Sydney warehouse, arriving Thursday",
    "A-2001": "processing, ships Friday",
}


@tool
def get_order_status(order_id: str) -> str:
    """Look up an order's fulfilment status by its id, for example A-1042."""
    if order_id not in ORDERS:
        return f"no order found for {order_id}"
    return ORDERS[order_id]


class SupportTicket(BaseModel):
    """A support ticket, classified into typed fields."""

    category: Literal["billing", "bug", "question"] = Field(
        description="billing, bug, or question"
    )
    urgency: int = Field(ge=1, le=5, description="1-5, 5 being most urgent")
    summary: str = Field(description="one sentence, no invented facts")
    order_id: str | None = Field(default=None, description="order id if mentioned")


# --- model wiring ----------------------------------------------------------


def missing_key_message() -> str:
    return (
        "DEEPSEEK_API_KEY is not set.\n"
        '  PowerShell (this session):  $env:DEEPSEEK_API_KEY = "sk-..."\n'
        '  PowerShell (persist):       setx DEEPSEEK_API_KEY "sk-..."   # reopen terminal\n'
        "  Get a key:                  https://platform.deepseek.com/api_keys"
    )


def build_model() -> Any:
    """Return a live ChatDeepSeek handle, or explain exactly what is missing."""
    if not os.getenv("DEEPSEEK_API_KEY"):
        raise RuntimeError(missing_key_message())

    from langchain_deepseek import ChatDeepSeek

    return ChatDeepSeek(
        model=os.getenv("DEEPSEEK_MODEL", MODEL),
        api_key=os.getenv("DEEPSEEK_API_KEY"),
        api_base=os.getenv("DEEPSEEK_BASE_URL", BASE_URL),
        temperature=0,
        max_retries=2,
        timeout=60,
        # DeepSeek thinks by default; this lab is about message plumbing,
        # so keep replies short and cheap. Delete this to watch it reason.
        extra_body={"thinking": {"type": "disabled"}},
    )


# --- output helpers --------------------------------------------------------


def describe(message: BaseMessage) -> str:
    text = message.content if isinstance(message.content, str) else str(message.content)
    return f"{message.type:<6} {text}"


def print_trace(messages: list[BaseMessage]) -> None:
    for index, message in enumerate(messages, start=1):
        print(f"  {index} {describe(message)}")


def describe_tool(tool_obj: BaseTool) -> str:
    schema = tool_obj.args_schema
    if hasattr(schema, "model_json_schema"):
        schema = schema.model_json_schema()
    required = schema.get("required", []) if isinstance(schema, dict) else []
    return f"{tool_obj.name}{required}"


# --- stages ----------------------------------------------------------------


def stage_chat(model: Any) -> list[BaseMessage]:
    print("== 1. chat: a list of messages ===========================")
    messages = [SystemMessage(content=SYSTEM), HumanMessage(content=QUESTION_1)]
    print(f"turn 1: {len(messages)} messages sent")
    print_trace(messages)
    reply = model.invoke(messages)
    print(f"reply 1: {reply.content}")

    messages = messages + [reply, HumanMessage(content=QUESTION_2)]
    print(f"turn 2: {len(messages)} messages sent (the whole history is re-sent)")
    print_trace(messages)
    reply = model.invoke(messages)
    print(f"reply 2: {reply.content}")
    print()
    return messages


def stage_stream(model: Any) -> None:
    print("== 2. streaming: the same reply, in pieces ===============")
    messages = [SystemMessage(content=SYSTEM), HumanMessage(content=QUESTION_1)]
    baseline = model.invoke(messages).content
    chunks = list(model.stream(messages))
    pieces = [str(chunk.content) for chunk in chunks if chunk.content]
    joined = "".join(pieces)
    print(f"chunks: {len(chunks)} ({len(pieces)} with text, {len(chunks) - len(pieces)} empty)")
    print(f"joined == invoke(): {joined == baseline}")
    print(f'first chunk: "{pieces[0]}"')
    print(f'last text chunk: "{pieces[-1]}"')
    print()


def stage_tools(model: Any) -> None:
    print("== 3. tools: one round trip, not an agent ================")
    print(f"tool schema: {describe_tool(get_order_status)}")
    bound = model.bind_tools([get_order_status])
    messages = [SystemMessage(content=SYSTEM), HumanMessage(content=QUESTION_3)]
    ai = bound.invoke(messages)
    call = ai.tool_calls[0]
    print(f"model requested: {call['name']}({call['args']})")

    result = get_order_status.invoke(call["args"])
    print(f"your code ran:   {result}")

    tool_message = ToolMessage(content=result, tool_call_id=call["id"])
    print(f"tool_call_id:    {call['id']} (matches the request)")

    thread = messages + [ai, tool_message]
    final = bound.invoke(thread)
    thread = thread + [final]
    print(f"final:           {final.content}")
    print(f"thread length:   {len(thread)} messages")
    print()


def stage_structured(model: Any) -> None:
    print("== 4. structured output: natural language in, pydantic ==")
    classifier = model.with_structured_output(SupportTicket, method="function_calling")
    ticket = classifier.invoke(
        [
            SystemMessage(content="Classify the support ticket. Extract the order id."),
            HumanMessage(content=TICKET_TEXT),
        ]
    )
    assert isinstance(ticket, SupportTicket)
    print(f"result: category={ticket.category!r} urgency={ticket.urgency} "
          f"order_id={ticket.order_id!r}")
    print(f"type:   {type(ticket).__name__} (validated instance)")
    print()


def run() -> int:
    try:
        model = build_model()
    except RuntimeError as error:
        print(f"error: {error}", file=sys.stderr)
        return 1

    print(f"DeepSeek via LangChain -- live model ({MODEL})")
    print()
    stage_chat(model)
    stage_stream(model)
    stage_tools(model)
    stage_structured(model)
    print("done: 4 stages, live DeepSeek calls")
    return 0


def main() -> int:
    try:
        return run()
    except Exception as error:  # noqa: BLE001 - surface the provider error, keep exit codes
        print(f"error: {type(error).__name__}: {error}", file=sys.stderr)
        print(
            "check DEEPSEEK_API_KEY, account balance, and network access.",
            file=sys.stderr,
        )
        return 1


if __name__ == "__main__":
    raise SystemExit(main())

evals.py

"""Self-tests for 0002-first-llm-api.

No fake model: the lab talks to live DeepSeek. These checks cover everything that
does not need the network (message shapes, tool schema, pydantic contract, model
wiring, the missing-key instructions), plus one live smoke call when a key is set.
"""

from __future__ import annotations

import os
import re
import subprocess
import sys
import tomllib
from pathlib import Path

import langchain_lab as lab
from langchain_core.messages import AIMessage, HumanMessage, SystemMessage

HERE = Path(__file__).resolve().parent
PY = sys.executable
CHECKS: list[str] = []


def check(name: str, condition: bool, detail: str = "") -> None:
    if not condition:
        raise AssertionError(f"{name}: {detail}" if detail else name)
    CHECKS.append(name)


def clean_env() -> dict[str, str]:
    env = {k: v for k, v in os.environ.items() if not k.startswith("DEEPSEEK_")}
    env.pop("OPENAI_API_KEY", None)
    return env


def run_lab(env: dict[str, str], timeout: int = 90):
    return subprocess.run(
        [PY, "langchain_lab.py"],
        cwd=HERE,
        env=env,
        capture_output=True,
        text=True,
        encoding="utf-8",
        errors="replace",
        timeout=timeout,
    )


def with_env(values: dict[str, str]):
    """Temporarily set env vars, then restore exactly."""

    class _Ctx:
        def __enter__(self):
            self.old = os.environ.copy()
            os.environ.update(values)

        def __exit__(self, *args):
            os.environ.clear()
            os.environ.update(self.old)

    return _Ctx()


# --- tests -----------------------------------------------------------------


def test_ticket_schema() -> None:
    schema = lab.SupportTicket.model_json_schema()
    props = schema["properties"]
    check("category is an enum", props["category"]["enum"] == ["billing", "bug", "question"])
    check("urgency has 1..5 bounds", props["urgency"]["minimum"] == 1 and props["urgency"]["maximum"] == 5)
    check("order_id is optional", "order_id" not in schema.get("required", []))
    check("summary required", "summary" in schema.get("required", []))


def test_ticket_validation() -> None:
    ticket = lab.SupportTicket(
        category="billing", urgency=5, summary="Duplicate charge.", order_id="A-1042"
    )
    check("valid ticket accepted", ticket.order_id == "A-1042")
    for bad in ({"category": "billing", "urgency": 9, "summary": "x"},
                {"category": "other", "urgency": 1, "summary": "x"}):
        try:
            lab.SupportTicket(**bad)
        except Exception:
            check(f"rejects {bad.get('urgency', bad.get('category'))}", True)
        else:
            check(f"rejects {bad}", False)


def test_tool_contract() -> None:
    schema = lab.get_order_status.args_schema.model_json_schema()
    check("tool name", lab.get_order_status.name == "get_order_status")
    check("tool takes order_id", schema.get("required") == ["order_id"])
    check("tool description mentions status", "status" in lab.get_order_status.description.lower())
    check("fixture returns a status", lab.get_order_status.invoke({"order_id": "A-1042"}).startswith("shipped"))
    check("fixture handles unknown ids", "no order found" in lab.get_order_status.invoke({"order_id": "X-1"}))


def test_message_protocol() -> None:
    messages = [SystemMessage(content=lab.SYSTEM), HumanMessage(content=lab.QUESTION_1)]
    check("turn 1 has 2 messages", len(messages) == 2)
    check("roles are system + human", [m.type for m in messages] == ["system", "human"])
    reply = AIMessage(content="Send us the invoice number.")
    thread = messages + [reply, HumanMessage(content=lab.QUESTION_2)]
    check("turn 2 has 4 messages", len(thread) == 4)
    check("history includes the AI reply", thread[2].type == "ai")


def test_model_config() -> None:
    with with_env({"DEEPSEEK_API_KEY": "test-key"}):
        model = lab.build_model()
    check("live model class", type(model).__name__ == "ChatDeepSeek", type(model).__name__)
    check("default model name", model.model_name == "deepseek-flash", model.model_name)
    check("default base url", str(model.root_client.base_url).rstrip("/") == "https://api.deepseek.com")
    check("temperature zero", model.temperature == 0)
    check(
        "thinking disabled for the lab",
        model.extra_body == {"thinking": {"type": "disabled"}},
        str(model.extra_body),
    )


def test_model_overrides() -> None:
    with with_env({
        "DEEPSEEK_API_KEY": "test-key",
        "DEEPSEEK_MODEL": "deepseek-v4-pro",
        "DEEPSEEK_BASE_URL": "https://example.test",
    }):
        model = lab.build_model()
    check("model override", model.model_name == "deepseek-v4-pro", model.model_name)
    check("base url override", str(model.root_client.base_url).rstrip("/") == "https://example.test")


def test_missing_key() -> None:
    result = run_lab(clean_env())
    check("no key exits 1", result.returncode == 1, str(result.returncode))
    check("shows session command", '$env:DEEPSEEK_API_KEY = "sk-..."' in result.stderr, result.stderr)
    check("shows persist command", "setx DEEPSEEK_API_KEY" in result.stderr, result.stderr)
    check("shows key page", "platform.deepseek.com/api_keys" in result.stderr, result.stderr)
    check("no banner before failing", "live model" not in result.stdout, result.stdout)


def test_no_fake_model() -> None:
    check("no FakeDeepSeek in the lab", not hasattr(lab, "FakeDeepSeek"))
    source = (HERE / "langchain_lab.py").read_text(encoding="utf-8")
    check("lab has no scripted branch", "--real" not in source and "scripted" not in source.lower())


def test_live_smoke() -> tuple[bool, str]:
    if not os.getenv("DEEPSEEK_API_KEY"):
        return False, "skipped (no DEEPSEEK_API_KEY)"
    model = lab.build_model()
    reply = model.invoke(
        [SystemMessage(content="Answer in one sentence."), HumanMessage(content="Say hello.")]
    )
    check("live reply is an AIMessage", isinstance(reply, AIMessage), type(reply).__name__)
    check("live reply has text", bool(str(reply.content).strip()))
    return True, "ran one live call"


def test_uv_project() -> None:
    """The uv project is the single source of dependencies; keep it honest."""
    pyproject = HERE / "pyproject.toml"
    check("pyproject.toml exists", pyproject.exists())
    data = tomllib.loads(pyproject.read_text(encoding="utf-8"))
    declared = {
        re.split(r"[<>=!~ ]", dep, maxsplit=1)[0]
        for dep in data["project"]["dependencies"]
    }
    check(
        "declares langchain, langchain-deepseek, pydantic",
        {"langchain", "langchain-deepseek", "pydantic"} <= declared,
        str(sorted(declared)),
    )


TESTS = [
    ("ticket schema", test_ticket_schema),
    ("ticket validation", test_ticket_validation),
    ("tool contract", test_tool_contract),
    ("message protocol", test_message_protocol),
    ("model config", test_model_config),
    ("model overrides", test_model_overrides),
    ("missing key instructions", test_missing_key),
    ("no fake model", test_no_fake_model),
    ("uv project", test_uv_project),
]


def main() -> int:
    failures = 0
    for index, (name, test) in enumerate(TESTS, start=1):
        try:
            test()
            print(f"ok {index:02d} {name}")
        except Exception as error:  # noqa: BLE001 - a test runner reports everything
            failures += 1
            print(f"FAIL {index:02d} {name}: {error}")

    try:
        ran, note = test_live_smoke()
    except Exception as error:  # noqa: BLE001
        failures += 1
        print(f"FAIL 10 live smoke: {error}")
    else:
        print(f"ok 10 live smoke: {note}")
        if not ran:
            print("     set DEEPSEEK_API_KEY to include the live call")

    total = len(CHECKS)
    print(f"\n{total} checks passed, {failures} failed")
    return 0 if failures == 0 else 1


if __name__ == "__main__":
    raise SystemExit(main())
